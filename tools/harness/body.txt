# A body that keeps its wounds (phase 5): a broken leg slows a walk 45 %, a
# splint brings that to 20 %; a deep arm cut opens again on a swing until it
# is stitched; a dressing left on 12 hours goes dirty; the BODY tab.
# Run in a fresh directory:
#   dead_pixel_survival.exe --script tools/harness/body.txt --hidden --world-seed 1234 --shots shots/body

newrun
god on
trait plain
waits 1
slay 80
zoom 2.5

# Five seconds walking along the street, whole: 900 px less the start, about
# 27.8 tiles (something on the street stops it a little short).
goto pavement
waits 0.5
mark
goto mark
waits 0.5
hold D 300
waits 5.5
expect moved 25.5 28.5

# A broken left leg: 45 % slower, 0.55 x 27.8 = 15.3 tiles.
goto mark
waits 0.5
wound fracture lleg
expect condition Broken leg
mark
hold D 300
waits 5.5
expect moved 14.6 16.0
shot 01_broken_leg_limp

# Splinted (ten seconds of standing still): 20 % slower, 0.8 x 27.8 = 22.2.
goto mark
waits 0.5
give Splint
treat Splint
waits 0.5
expect action busy
waits 10.5
expect action none
expect wounds 1 splinted
mark
hold D 300
waits 5.5
expect moved 21.5 22.9
