# Facing: characters turn on screen toward where they aim or walk, with a back
# view going up the screen and a mirrored one for the left.

#   dead_pixel_survival.exe --script tools/harness/facing.txt --hidden --world-seed 1234 --shots shots/facing

god on

zoom 3
mouse 900 520
waits 0.3
shot 01_aim_down_right
mouse 380 520
waits 0.3
shot 02_aim_down_left

waits 0.3
shot 03_aim_up_right

waits 0.3

hold W 20

shot 06_crouch

shot 07_crouch_walk

# Zombies closing in from three sides of the room.

shot 08_zombies_coming

shot 09_close
