# The dead in numbers (phase 4): 300 of them around the player, 600 ticks
# timed. Average tick under 4 ms, worst under 12 ms (Release build). Then
# 20 survivors and 100 of the dead, the same budget.
# Run in a fresh directory:
#   dead_pixel_survival.exe --script tools/harness/perf.txt --hidden --world-seed 1234 --shots shots/perf

mode swarm
newrun
god on
waits 1
hours 22
slay 200
# The Sergeant is among the slain since phase 9. His key no longer writes the
# gate file in the tick he falls (phase 10), so the timing starts at once.
perf 300
waits 2
shot perf_crowd
waits 9
expect perf 4 12

# The living too (phase 8): 20 survivors around the player, by day, and
# 100 of the dead.
slay 400
hours 12
survivor crowd 20
waits 1
perf 100
waits 2
shot perf_survivors
waits 9
expect perf 4 12
survivors

# Vehicles (phase 11): the town's ten stand parked through the timings
# above. Then a sedan driven up the road through 300 of the dead.
slay 400
survivors gone
hours 22
goto vehicle sedan
set key in
set fuel 30
set condition 100
tap V
waits 0.3
tap E
waits 0.5
expect vehicle engine
god on
perf 300
drive road 11
shot perf_driving
expect perf 4 12
vehicles
quit
